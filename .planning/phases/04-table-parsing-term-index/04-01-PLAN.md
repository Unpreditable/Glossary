```typescript
import { App, TFile } from "obsidian";

const GLOSSARY_TAG = "glossary" as const;

export function isGlossaryFile(app: App, file: TFile): boolean {
  const cache = app.metadataCache.getFileCache(file);
  const tags = cache?.frontmatter?.tags as string | string[] | undefined;
  return Array.isArray(tags)
    ? tags.includes(GLOSSARY_TAG)
    : tags === GLOSSARY_TAG;
}
```

The file must export three functions:

**addToIndex(key, term, termIndex):** Pushes term into the array at termIndex[key], creating the array if it doesn't exist. Pure helper.

**isExcluded(term, type, settings):** Checks term against excludeAcronyms (type="acronym") or excludeTerms (type="term"). Uses caseSensitive for comparison — exact match when true, toLowerCase() comparison when false. Source from RESEARCH.md Pattern 5.

**parseGlossaryFile(app, file, settings, termIndex):** Returns Promise<number> (count of rows parsed — NOT index keys written). Algorithm:
1. Read file content via `app.vault.read(file)`
2. Split by "\n", iterate lines
3. Skip lines not starting and ending with "|"
4. Skip separator rows: `line.match(/^\|[\s\-\|]+\|$/)` — use this exact regex
5. Skip header rows: lookahead checks if next line is a separator — same regex applied to `lines[i + 1].trim()`
6. Extract columns: `line.split("|").map(col => col.trim()).slice(1, -1)`
7. Skip if column count is not 2 or 3; skip if all columns are empty
8. For 3-col rows: extract [acronym, fullTerm, description]. Apply matchBy gating — see RESEARCH.md full algorithm. Apply isExcluded() before writing. Key fullTerm by `settings.caseSensitive ? fullTerm : fullTerm.toLowerCase()`.
9. For 2-col rows: extract [acronym, description]. Always write acronym regardless of matchBy. Apply isExcluded() for acronym.
10. For every acronym added to index (both 2-col and 3-col): also call `addToIndex(acronym + "s", term, termIndex)`. The plural shares the same term object. The plural add does NOT increment count.
11. Increment count only when `added = true` (a row produced at least one index write).
12. Return count.

The complete reference implementation is in RESEARCH.md section "Full parseGlossaryFile() Algorithm" — port it directly.
